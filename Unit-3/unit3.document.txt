Q7.Write a program to copy move and delete files using shutil module. 


import shutil
import os

# Create a sample file
with open("sample.txt", "w") as file:
    file.write("Hello, Python!")

# 1. Copy the file
shutil.copy("sample.txt", "copy.txt")
print("File copied successfully.")

# 2. Move the file
shutil.move("copy.txt", "moved.txt")
print("File moved successfully.")

# 3. Delete the file
if os.path.exists("moved.txt"):
    os.remove("moved.txt")
    print("File deleted successfully.")

Q8.Write a program to demonstrate basic regular expression pattern matching. 

import re

text = "My name is Arjun and my phone number is 9876543210."

# Search for a word
result = re.search("Arjun", text)

if result:
    print("Word found:", result.group())
else:
    print("Word not found.")

# Find all digits
digits = re.findall(r"\d", text)
print("Digits:", digits)

# Check whether the text starts with a word
result = re.match("My", text)

if result:
    print("Text starts with:", result.group())
else:
    print("No match at the beginning.")



Q9.Write a program to use re module functions such as match search and findall.

import re

text = "Python is easy. Python is powerful. My marks are 85."

# 1. Using match()
result1 = re.match("Python", text)
print("Match:", result1.group() if result1 else "Not found")

# 2. Using search()
result2 = re.search("easy", text)
print("Search:", result2.group() if result2 else "Not found")

# 3. Using findall()
result3 = re.findall("Python", text)
print("Findall:", result3)

# Find all digits
digits = re.findall(r"\d+", text)
print("Digits:", digits)


Q10.#10.Write a program to extract specific information from a text file using regular expressions


import re

# Create a text file
with open("data.txt", "w") as file:
    file.write("Name: Arjun\n")
    file.write("Phone: 9876543210\n")
    file.write("Email: arjun@gmail.com\n")

# Read the text file
with open("data.txt", "r") as file:
    text = file.read()

# Extract name
name = re.search(r"Name:\s*(.*)", text)

# Extract phone number
phone = re.search(r"Phone:\s*(\d{10})", text)

# Extract email
email = re.search(r"Email:\s*([\w.-]+@[\w.-]+\.\w+)", text)

# Display extracted information
print("Name:", name.group(1) if name else "Not found")
print("Phone:", phone.group(1) if phone else "Not found")
print("Email:", email.group(1) if email else "Not found")
